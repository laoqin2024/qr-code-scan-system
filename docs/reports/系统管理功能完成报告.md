router.post('/batch-delete-invalid', requireAuth, async (req: AuthRequest, res) => {
  // 构建删除条件
  let query = 'DELETE FROM scans WHERE is_valid = 0';
  
  if (customer_id) query += ' AND customer_id = ?';
  if (product_id) query += ' AND product_id = ?';
  if (start_time) query += ' AND created_at >= ?';
  if (end_time) query += ' AND created_at <= ?';

  const result = db.prepare(query).run(...params);
  
  // 返回删除数量
  res.json({ deleted_count: result.changes });
});
```

    // 3. 删除产品权限
    // 4. 删除产品
    // 5. 删除客户
    // 6. 删除用户权限
    // 7. 删除用户
    
    // 提交事务
    db.exec('COMMIT');
    db.exec('PRAGMA foreign_keys = ON');
  } catch (error) {
    // 回滚
    db.exec('ROLLBACK');
    db.exec('PRAGMA foreign_keys = ON');
    throw error;
  }
});
```

**关键技术：**
- 使用事务确保原子性
- 临时禁用外键检查避免循环引用
- 按正确顺序删除数据
- 失败自动回滚

## 📝 数据库影响

### 删除顺序

```
1. 扫码记录 (scans)
   ↓
2. 产品权限 (user_product_permissions)
   ↓
3. 产品 (products)
   ↓
4. 客户 (customers)
   ↓
5. 用户权限 (user_product_permissions)
   ↓
6. 用户 (users)
```

### 外键处理

**问题：** 循环引用
- `customers.created_by` → `users.id`
- `users.customer_id` → `customers.id`

**解决方案：**
- 临时禁用外键检查
- 使用事务确保一致性
- 按正确顺序删除
